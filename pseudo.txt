# FUNCTION 1 - MASS REQUIREMENT TEST
# ==========================================================

DEFINE FUNCTION run_mass_test(vehicle):

    PRINT "MASS REQUIREMENT TEST"

    SET regulated_mass, total_mass =
        CALL calculate_mass(vehicle)

    SET margin = MAX_MASS - regulated_mass

    IF regulated_mass <= MAX_MASS:
        SET status = "PASS"
    ELSE:
        SET status = "FAIL"

    PRINT status
    PRINT regulated mass
    PRINT mass margin

    SET result = DICTIONARY:
        "test": "Mass Requirement"
        "status": status
        "measured": regulated_mass
        "limit": MAX_MASS
        "unit": "g"

    RETURN result

# ==========================================================
# FUNCTION 6 - COMPONENT DATA CHECK
# ==========================================================

DEFINE FUNCTION run_component_check(vehicle):

    PRINT "COMPONENT DATA VERIFICATION"

    SET errors = EMPTY LIST
    SET component_ids = EMPTY SET

    FOR each component IN vehicle["components"]:

        IF component ID is missing or empty:
            ADD error to errors

        ELSE IF component ID already in component_ids:
            ADD duplicate ID error to errors

        ELSE:
            ADD component ID to component_ids

        CHECK required fields:
            id
            name
            category
            mass_g
            cost_gils
            position_mm
            excl_status

        IF any required field missing:
            ADD error to errors
            CONTINUE to next component

        IF mass is negative:
            ADD error to errors

        IF cost is negative:
            ADD error to errors

        IF position_mm does not contain 3 numbers:
            ADD error to errors

        IF exclusion status is not Boolean:
            ADD error to errors

    IF errors list is empty:
        SET status = "PASS"
    ELSE:
        SET status = "FAIL"

    PRINT status
    PRINT number of errors

    FOR each error IN errors:
        PRINT error

    RETURN DICTIONARY:
        "test": "Component Data Check"
        "status": status
        "errors": errors


# ==========================================================
# FUNCTION 7 - FULL TEST SUITE
# ==========================================================

DEFINE FUNCTION run_test_suite(vehicle):

    PRINT "LTA McQueen - Verification Suite"

    SET results = EMPTY LIST

    # Validate component data before calculating
    # mass and cost.

    SET component_result =
        CALL run_component_check(vehicle)

    ADD component_result TO results

    IF component_result status == "FAIL":
        PRINT "Fix component data before running"
        PRINT "the remaining tests."
        RETURN results

    SET mass_result = CALL run_mass_test(vehicle)
    ADD mass_result TO results

    SET budget_result = CALL run_budget_test(vehicle)
    ADD budget_result TO results

    # Additional performance tests require
    # input values from the user.

    ASK "Run lift test? (Y/N)"

    IF answer == "Y":
        SET lift_result = CALL run_lift_test(vehicle)

        IF lift_result IS NOT None:
            ADD lift_result TO results

    ASK "Run thrust test? (Y/N)"

    IF answer == "Y":
        SET thrust_result = CALL run_thrust_test()

        IF thrust_result IS NOT None:
            ADD thrust_result TO results

    ASK "Run endurance test? (Y/N)"

    IF answer == "Y":
        SET endurance_result = CALL run_endurance_test()

        IF endurance_result IS NOT None:
            ADD endurance_result TO results

    # Count results

    SET passed = 0
    SET failed = 0

    FOR each result IN results:

        IF result["status"] == "PASS":
            INCREASE passed BY 1

        ELSE IF result["status"] == "FAIL":
            INCREASE failed BY 1

    PRINT "================================"
    PRINT "VERIFICATION SUMMARY"
    PRINT "================================"
    PRINT "Tests Run: " + LENGTH of results
    PRINT "Passed: " + passed
    PRINT "Failed: " + failed

    RETURN results

# FUNCTION 8 - SAVE TEST REPORT
# ==========================================================

DEFINE FUNCTION save_test_report(vehicle, results):

    PRINT "SAVE TEST REPORT"

    IF results is empty:
        PRINT "No test results to save."
        RETURN False

    GET current date and time

    CREATE timestamp string suitable for filenames

    SET report_filename =
        vehicle revision + "_" + timestamp + ".json"

    SET report_path = REPORT_DIR / report_filename

    CREATE report DICTIONARY:
        "vehicle": vehicle["name"]
        "revision": vehicle["revision"]
        "timestamp": current date and time
        "results": results

    TRY:
        OPEN report_path in write mode

        SAVE report dictionary using json.dump
        with indentation of 4

    EXCEPT file writing error:
        PRINT "Unable to save report."
        RETURN False

    PRINT "Test report saved successfully."
    PRINT report_path

    RETURN True
