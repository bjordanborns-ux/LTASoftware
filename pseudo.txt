DEFINE FUNCTION physics_menu(vehicle):

    WHILE True:

        PRINT "================================"
        PRINT "LTA McQueen - Physics Analysis"
        PRINT "================================"
        PRINT "1 - Calculate Balloon Volume"
        PRINT "2 - Calculate Buoyant Force"
        PRINT "3 - Calculate Vehicle Weight"
        PRINT "4 - Calculate Net Lift"
        PRINT "5 - Calculate Aerodynamic Drag"
        PRINT "6 - Calculate Thrust-to-Weight Ratio"
        PRINT "7 - Calculate Acceleration"
        PRINT "8 - Estimate Flight Time"
        PRINT "9 - Return to Main Menu"

        SET selection = INPUT "Select option 1-9: "
        REMOVE whitespace from selection

        IF selection == "1":
            CALL calculate_volume()

        ELSE IF selection == "2":
            SET volume = calculate_volume()

            IF volume IS NOT None:
                CALL calculate_buoyancy(volume)

        ELSE IF selection == "3":
            CALL calculate_weight(vehicle)

        ELSE IF selection == "4":
            SET volume = calculate_volume()

            IF volume IS NOT None:
                CALL calculate_net_lift(vehicle, volume)

        ELSE IF selection == "5":
            CALL calculate_drag()

        ELSE IF selection == "6":
            CALL calculate_thrust_to_weight(vehicle)

        ELSE IF selection == "7":
            CALL calculate_acceleration(vehicle)

        ELSE IF selection == "8":
            CALL calculate_flight_time()

        ELSE IF selection == "9":
            BREAK OUT OF WHILE LOOP

        ELSE:
            PRINT "Error: Select 1-9"

ELSE IF choice is main menu: 
    main menu code

ELSE: 
    Must choose main or physics.
