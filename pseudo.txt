# IMPORTS
IMPORT the following functions FROM vehicle.py:
    load_vehicle
    add_component
    edit_component
    remove_component
    view_components
    calculate_mass
    calculate_cost
    new_revision
    view_revisions


# ==================================================
# MAIN FUNCTION
# ==================================================

DEFINE FUNCTION main():

    # STEP 1 - LOAD VEHICLE

    SET vehicle = CALL load_vehicle()

    IF vehicle IS None:
        PRINT "ERROR: Unable to load vehicle."
        PRINT "Application startup aborted."
        RETURN


    # STEP 2 - MAIN MENU LOOP

    WHILE True:

        PRINT blank line
        PRINT "================================"
        PRINT "       LTA McQueen System"
        PRINT "================================"
        PRINT "Vehicle: " + vehicle name
        PRINT "Active Revision: " + vehicle revision
        PRINT "--------------------------------"
        PRINT "1 - View Components"
        PRINT "2 - Register New Component"
        PRINT "3 - Edit Component"
        PRINT "4 - Remove Component"
        PRINT "5 - Calculate Mass"
        PRINT "6 - Calculate Cost"
        PRINT "7 - Finalize Revision"
        PRINT "8 - View Revision History"
        PRINT "9 - Exit"
        PRINT "================================"

        # STEP 3 - GET MENU SELECTION

        SET selection = INPUT "Select option (1-9): "

        REMOVE whitespace from selection


        # STEP 4 - HANDLE MENU SELECTION

        IF selection EQUALS "1":
            CALL view_components(vehicle)

        ELSE IF selection EQUALS "2":
            CALL add_component(vehicle)

        ELSE IF selection EQUALS "3":
            CALL edit_component(vehicle)

        ELSE IF selection EQUALS "4":
            CALL remove_component(vehicle)

        ELSE IF selection EQUALS "5":
            CALL calculate_mass(vehicle)

        ELSE IF selection EQUALS "6":
            CALL calculate_cost(vehicle)

        ELSE IF selection EQUALS "7":
            CALL new_revision(vehicle)

        ELSE IF selection EQUALS "8":
            CALL view_revisions()

        ELSE IF selection EQUALS "9":
            PRINT "Exiting LTA McQueen System..."
            BREAK OUT OF WHILE LOOP

        ELSE:
            PRINT "ERROR: Invalid selection."
            PRINT "Please enter a number from 1 to 9."


# ==================================================
# PROGRAM ENTRY POINT
# ==================================================

IF this file is being executed directly:
    CALL main()
